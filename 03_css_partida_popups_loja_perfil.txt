// Obter arquétipos disponíveis para o grupo de posição
function getArquetipoPorGrupo(grupo) {
  return ARQUETIPOS[grupo] || ARQUETIPOS.ATA;
}

// Aplicar bônus do arquétipo nos attrs
function applyArquetipoBonus(arqKey, grupo) {
  const arqs = getArquetipoPorGrupo(grupo);
  const arq  = arqs.find(a => a.key === arqKey);
  if (!arq) return;
  Object.entries(arq.bnus).forEach(([stat, val]) => {
    addAttrBonus(stat, val);
  });
}

// Aplicar bônus de um talento
function applyTalentoBonus(talKey, grupo) {
  const arqs = getArquetipoPorGrupo(grupo);
  for (const arq of arqs) {
    const tal = arq.talentos?.find(t => t.key === talKey);
    if (tal && tal.attrBonus) {
      Object.entries(tal.attrBonus).forEach(([stat, val]) => {
        addAttrBonus(stat, val);
      });
      return tal;
    }
  }
  return null;
}

const ATTR_ALIAS = {
  defesa: 'tackle',
  posicionamento: 'posicao',
  mental: 'reacoes',
  fisico: 'forca'
};

function attrKey(key) {
  return ATTR_ALIAS[key] || key;
}

function addAttrBonus(key, val) {
  const realKey = attrKey(key);
  G.attrs[realKey] = Math.min(99, (G.attrs[realKey]||0) + val);
}

function getSkillValue(key, fallback = 0) {
  if (!G || !G.attrs) return fallback;
  if (key === 'fisico') {
    return Math.round(((G.attrs.forca||0) + (G.attrs.resistencia||0) + (G.attrs.salto||0)) / 3) || fallback;
  }
  if (key === 'mental') {
    return Math.round(((G.attrs.reacoes||0) + (G.attrs.posicao||0) + (G.compostura||5)) / 3) || fallback;
  }
  const realKey = attrKey(key);
  return G.attrs[realKey] ?? fallback;
}
