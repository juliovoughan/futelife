function proximoAdversarioClubeTabela(clube) {
  if (!G || !clube) return '--';
  if (clube === G.clube) return getProxPartida()?.adversario || '--';
  const t = initTabelaLiga(false);
  const inicio = Math.max((t?.rodadaAtual||0) + 1, G.rodada||1);
  const max = Math.min(inicio + 5, 38);
  for (let rodada = inicio; rodada <= max; rodada++) {
    const jogo = gerarParesRodadaLiga(rodada).find(p => p.home === clube || p.away === clube);
    if (jogo) return jogo.home === clube ? jogo.away : jogo.home;
  }
  return '--';
}

function tabelaLigaHtml(limit = 14) {
  const rows = ordenarTabelaLiga().slice(0, limit);
  if (!rows.length) return '<div class="empty-notice">Tabela ainda sem jogos.</div>';
  return '<div class="league-table-wrap"><table class="league-table">'+
    '<thead><tr><th>#</th><th>Clube</th><th>Pts</th><th>J</th><th>V</th><th>E</th><th>D</th><th>SG</th><th>Proximo</th></tr></thead><tbody>'+
    rows.map((r, i) => '<tr class="'+(r.clube===G.clube?'my-row':'')+'">'+
      '<td>'+(i+1)+'</td>'+
      '<td class="club-cell">'+r.clube+'</td>'+
      '<td>'+r.pts+'</td>'+
      '<td>'+r.j+'</td>'+
      '<td>'+r.g+'</td>'+
      '<td>'+r.e+'</td>'+
      '<td>'+r.d+'</td>'+
      '<td>'+r.sg+'</td>'+
      '<td class="next-cell">'+proximoAdversarioClubeTabela(r.clube)+'</td>'+
    '</tr>').join('')+
    '</tbody></table></div>';
}

function setModoSimGlobal(modo) {
  CONFIG_SIM.modo = modo;
  if (modo === 'todas') CONFIG_SIM.ligasAtivas = Object.keys(GLOBAL_BASES);
  salvarConfigSim();
  closePopup();
  openConfigSimGlobal();
}

function selecionarTodasLigasSimGlobal() {
  CONFIG_SIM.modo = 'selecionadas';
  CONFIG_SIM.ligasAtivas = Object.keys(GLOBAL_BASES);
  salvarConfigSim();
  closePopup();
  openConfigSimGlobal();
}

function limparLigasSimGlobal() {
  CONFIG_SIM.modo = 'selecionadas';
  CONFIG_SIM.ligasAtivas = [];
  salvarConfigSim();
  closePopup();
  openConfigSimGlobal();
}

function toggleLigaSimGlobal(id) {
  const set = new Set(CONFIG_SIM.ligasAtivas || []);
  if (set.has(id)) set.delete(id); else set.add(id);
  CONFIG_SIM.ligasAtivas = [...set];
  CONFIG_SIM.modo = 'selecionadas';
  salvarConfigSim();
  closePopup();
  openConfigSimGlobal();
}

function gerarObjetivoDiretoriaTemporada() {
  if (!G || G.perfil !== 'treinador' || G.promessaEpoca) return;
  initTabelaLiga(false);
  const ovr = clubeOvrEstimado(G.clube);
  let meta = 'Terminar na metade superior';
  if (ovr >= 78) meta = 'Ganhar o titulo';
  else if (ovr >= 72) meta = 'Top 3 no campeonato';
  else if (ovr <= 63) meta = 'Evitar rebaixamento';
  G.promessaEpoca = { meta, prazoAno:G.ano, cumprida:false, auto:true };
}

function avaliarObjetivoDiretoriaFimTemporada() {
  if (!G || G.perfil !== 'treinador') return;
  const p = G.promessaEpoca;
  if (!p) return;
  const status = objetivoDiretoriaTexto();
  p.cumprida = status === 'cumprido';
  G.ultimoObjetivoDiretoriaCumprido = p.cumprida;
  if (p.cumprida) {
    ajustarStatusClube(8, 8, 12);
    G.fama = Math.min(100, G.fama + 5);
    G.moral = Math.min(100, G.moral + 8);
    addNoticia(G,'OK','Objetivo da diretoria cumprido: '+p.meta+'.');
  } else {
    ajustarStatusClube(-12, -12, -22);
    G.fama = Math.max(0, G.fama - 10);
    G.moral = Math.max(0, G.moral - 14);
    addNoticia(G,'DIR','Objetivo da diretoria falhou: '+p.meta+'.');
    if ((G.clubeStatus?.apoioDiretoria||0) < 25) demitirTreinador('objetivos nao cumpridos');
  }
  G.promessaEpoca = null;
}

function demitirTreinador(motivo = 'pressao interna') {
  if (!G || G.perfil !== 'treinador') return;
  const clubes = [...(LIGAS[G.liga]?.clubes || CLUBES_PEQUENOS)].filter(c => c && c !== G.clube);
  const destino = clubes.slice(-6)[rng(0, Math.max(0, Math.min(5, clubes.length-1)))] || pickRandom(CLUBES_PEQUENOS);
  const antigo = G.clube;
  G.clube = destino;
  G.fama = Math.max(0, G.fama - 8);
  G.moral = Math.max(20, G.moral - 10);
  G.clubeStatus = { confiancaGrupo:48, apoioTorcida:45, apoioDiretoria:50, derrotasSeguidas:0, vitoriasSeguidas:0 };
  G.taticaTreinador = { formacao:'4-3-3', estilo:'equilibrado' };
  G.elencoClube = gerarElencoClubePersistente(G.clube, G.liga);
  G.orcamentoClube = calcularOrcamentoClube();
  G.mercadoTreinador = { chave:null, candidatos:[] };
  G.movsPlantel = { chave:null, usados:0 };
  initTabelaLiga(true);
  addNoticia(G,'DIR',`Demitido do ${antigo} por ${motivo}. Novo desafio: ${destino}.`);
}

const FORMACOES_TREINADOR = ['4-3-3','4-4-2','4-2-3-1','3-5-2','5-3-2'];

function normalizarTaticaTreinador() {
  if (!G.taticaTreinador || typeof G.taticaTreinador !== 'object') G.taticaTreinador = { formacao:G.formacao||'4-3-3', estilo:'equilibrado' };
  if (!FORMACOES_TREINADOR.includes(G.taticaTreinador.formacao)) G.taticaTreinador.formacao = G.formacao || '4-3-3';
  if (!TATICAS_TREINADOR[G.taticaTreinador.estilo]) G.taticaTreinador.estilo = 'equilibrado';
  G.formacao = G.taticaTreinador.formacao;
}

function setEstiloTreinador(estilo) {
  normalizarTaticaTreinador();
  if (!TATICAS_TREINADOR[estilo]) return;
  G.taticaTreinador.estilo = estilo;
  saveGame();
  closePopup();
  setTimeout(openTaticaTreinador, 0);
}

function setFormacaoTreinador(formacao) {
  normalizarTaticaTreinador();
  if (!FORMACOES_TREINADOR.includes(formacao)) return;
  G.taticaTreinador.formacao = formacao;
  G.formacao = formacao;
  saveGame();
  closePopup();
  setTimeout(openTaticaTreinador, 0);
}

function openTaticaTreinador() {
  if (!G || G.perfil !== 'treinador') { showToast('Tatica avancada disponivel no modo treinador.'); return; }
  normalizarTaticaTreinador();
  const atual = TATICAS_TREINADOR[G.taticaTreinador.estilo];
  const botoes = [
    ...FORMACOES_TREINADOR.map(f => ({ icon:f===G.taticaTreinador.formacao?'OK':'FOR', text:f, acao:()=>setFormacaoTreinador(f) })),
    ...Object.keys(TATICAS_TREINADOR).map(k => ({ icon:k===G.taticaTreinador.estilo?'OK':'TAC', text:TATICAS_TREINADOR[k].nome, acao:()=>setEstiloTreinador(k) })),
    { icon:'OK', text:'Fechar', acao:()=>{ window._popupQueue = []; closePopup(); } }
  ];
  showPopup({
    icon:'TAC',
    titulo:'Tatica do Treinador',
    corpo:'Formacao: '+G.taticaTreinador.formacao+
      '\nEstilo: '+atual.nome+
      '\n\n'+atual.desc+
      '\nEnergia: x'+atual.energiaDrain.toFixed(2)+
      '\nMods: A '+atual.mods.ataque+' / M '+atual.mods.meio+' / D '+atual.mods.defesa,
    botoes
  });
}

function aplicarTaticaTreinador() {
  if (!MS || !G || G.perfil !== 'treinador') return;
  normalizarTaticaTreinador();
  const t = TATICAS_TREINADOR[G.taticaTreinador.estilo] || TATICAS_TREINADOR.equilibrado;
  addMatchMod('ataque', t.mods.ataque || 0, 'tatica treinador');
  addMatchMod('meio', t.mods.meio || 0, 'tatica treinador');
  addMatchMod('defesa', t.mods.defesa || 0, 'tatica treinador');
  MS._taticaEnergiaMult = t.energiaDrain || 1;
  MS._taticaAplicada = true;
  logMatch('Tatica: '+G.taticaTreinador.formacao+' / '+t.nome+'.', 'hl');
}

function fecharNegociacao() {
  const ov = document.getElementById('popup-overlay');
  if (ov) ov.classList.remove('active');
  const el = document.getElementById('negoc-overlay');
  if (el) el.remove();
  _negocAtual = null;
}

function aceitarContrato() {
  if (!_negocAtual) return;
  const p = _negocAtual;
  const eraOutroClube = p.clube !== G.clube;

  // Comissão do agente: paga só uma vez, sobre 1 mês de salário (não o ano todo)
  // Isso evita dívida imediata de centenas de milhares
  const agente = AGENTES[G.tipoAgente||'nenhum'];
  const comissaoUnica = Math.round(p.salario * agente.comissao * 4); // 4 semanas
  if (comissaoUnica > 0) {
    if (G.dinheiro < comissaoUnica) {
      showToast('Comissao do agente insuficiente: precisa de '+fmtMoney(comissaoUnica));
      return;
    }
    showToast('Comissao do agente: -'+fmtMoney(comissaoUnica));
    G.dinheiro -= comissaoUnica;
    updateTopBar();
  }
  // Registrar contrato
  if (G.contrato) G.historico_contratos.push(G.contrato);
  G.contrato = { ...p };
  G.salario   = p.salario;
  G.contratoFim = G.ano + p.anos;
  G.tipoContrato = p.tipo || (G.perfil === 'treinador' ? 'treinador' : 'principal');
  G.semContratoDesde = null;
  // Transferência
  if (eraOutroClube) {
    const clubeAntigo = G.clube;
    G.clube = p.clube; G.liga = p.liga;
    G.jogosSeguidos = 0; G.entrosamento = Math.max(20, (G.entrosamento||30)-20);
    G.objetivoAtivo = null;
    G.rodadaLigaSimulada = 0;
    G.resultadosLiga = {};
    G.tabelaLiga = null;
    G.clubeStatus = { confiancaGrupo:55, apoioTorcida:55, apoioDiretoria:55, derrotasSeguidas:0, vitoriasSeguidas:0 };
    if (G.perfil === 'treinador') {
      normalizarLicencaTreinador();
      G.elencoClube = gerarElencoClubePersistente(G.clube, G.liga);
      G.orcamentoClube = calcularOrcamentoClube();
      G.mercadoTreinador = { chave:null, candidatos:[] };
      G.movsPlantel = { chave:null, usados:0 };
      refreshOVR();
    }
    addNoticia(G,'✈️',`Transferência! ${G.nome} assina com ${p.clube} por ${p.anos} ano${p.anos>1?'s':''}.`);
    showToast('✈️ Bem-vindo ao '+p.clube+'!');
  } else {
    addNoticia(G,'📋',`${G.nome} renova com ${p.clube} por ${p.anos} ano${p.anos>1?'s':''}.`);
    showToast('📋 Contrato renovado com '+p.clube+'!');
  }
  fecharNegociacao();
  saveGame(false); updateTopBar(); renderAgenteScreen();
}

  showPopup({

    corpo:`Você pede ${fmtMoney(pede)}/semana (+20%).

O clube tem paciência ${p.paciencia}/3.`,
    botoes:[
      { icon:'💰', text:`Pedir ${fmtMoney(pede)}/semana`, acao:()=>{ closePopup(); resolverContraproposta(pede, 0); }},
      { icon:'📅', text:'Pedir +1 ano de contrato',        acao:()=>{ closePopup(); resolverContraproposta(p.salario, 1); }},
      { icon:'🔒', text:'Pedir cláusula menor (-30%)',      acao:()=>{ closePopup(); resolverContraproposta(p.salario, 0, true); }},
    ]
  });
}

function resolverContraproposta(salPedido, anosPedidos, clausulaMenor) {
  if (!_negocAtual) return;
  const p = _negocAtual;
  p.paciencia--;
  const roll = Math.random();
  // Chance de aceitar baseada em OVR, fama e paciência
  const aceitaChance = Math.min(0.7, (G.ovr||50)/100 + (G.fama||5)/200);
  if (roll < aceitaChance * 0.6) {
    // Aceitar pedido
    if (salPedido > p.salario) p.salario = salPedido;
    if (anosPedidos > 0) p.anos = Math.min(5, p.anos + anosPedidos);
    if (clausulaMenor) p.clausula = Math.round(p.clausula * 0.70 / 100000) * 100000;
    showToast('✅ '+p.clube+' aceitou a contraproposta!');
    renderNegociacao(p);
  } else if (p.paciencia <= 0 || roll > 0.85) {
    // Clube irritado — encerra negociação
    showToast('😤 '+p.clube+' encerrou a negociação!');
    G.fama = Math.max(0, G.fama - 2);
    addNoticia(G,'😤','Negociação com '+p.clube+' fracassada. Fama ↓');
    fecharNegociacao();
    saveGame(false);
  } else {
    // Meio-termo: clube oferece metade da diferença
    const meio = Math.round((salPedido + p.salario) / 2 / 500) * 500;
    p.salario = meio;
    showToast('🤝 '+p.clube+' oferece meio-termo: '+fmtMoney(meio)+'/sem');
    renderNegociacao(p);
  }
}

function recusarProposta() {
  if (!_negocAtual) return;
  const clube = _negocAtual.clube;
  addNoticia(G,'❌',G.nome+' recusou proposta do '+clube+'.');
  fecharNegociacao();
}

// Gerar "furo" do agente (sondagem)
function gerarFuroAgente() {
  if (!G || G.perfil==='treinador') return;
  const tipo = G.tipoAgente||'nenhum';
  if (tipo==='nenhum') return;
  const mercado = gerarClubeMercado();
  const { clube, liga, score } = mercado;
  if ((G.furos||[]).some(f=>f.clube===clube&&f.ativo) || (G.rumores||[]).some(r=>r.clube===clube&&r.ativo&&r.ano===G.ano)) return;

  if (!isJanelaTransferencia() && !isPasseLivre()) {
    const rumor = gerarRumorTransferencia(clube, liga, score);
    showPopup({
      icon:'RUMOR', titulo:'Rumor de Mercado',
      corpo:'A imprensa diz que '+clube+' ('+liga+') esta observando '+G.nome+'.\n\nMoral +'+rumor.impactoMoral+' | Fama +'+rumor.impactoFama+(rumor.impactoEntrosamento?' | Entrosamento '+rumor.impactoEntrosamento:''),
      botoes:[{ icon:'OK', text:'Continuar', acao:()=>{ closePopup(); saveGame(false); renderAgenteScreen(); }}]
    });
    return rumor;
  }

  const prazo = 4 + Math.floor(Math.random()*3);
  const objetivo = gerarObjetivo(G.grupoPos, prazo, clube, liga);
  const furo = { clube, liga, objetivo, ativo:true, prazo, semanaInicio:G.semanaAtual, janela:isJanelaTransferencia(), livre:isPasseLivre() };
  G.furos = [...(G.furos||[]), furo];
  G.objetivoAtivo = objetivo;
  addNoticia(G,'SOND','Sondagem: '+clube+' ('+liga+') quer avaliar '+G.nome+'. Objetivo: '+objetivo.descricao);
  showPopup({
    icon:'SOND', titulo:isPasseLivre()?'Prova de Mercado':'Sondagem!',
    corpo:'Seu agente traz uma noticia:\n\n"'+clube+' esta a procura de um reforco."\n\nObjetivo para garantir a proposta:\n'+objetivo.descricao+'\n\nPrazo: '+prazo+' partidas',
    botoes:[
      { icon:'OK', text:'Aceitar o desafio!', acao:()=>{ closePopup(); saveGame(false); navTo('agente'); }},
      { icon:'X', text:'Ignorar por agora', acao:()=>{ G.objetivoAtivo=null; G.furos=G.furos.filter(f=>f.clube!==clube); closePopup(); }}
    ]
  });
  return furo;
}

function renderAgenteScreen() {
  if (!G) return;
  renderAgenteType();
  renderAgenteContrato();
  renderAgenteObjetivo();

  renderAgenteRumores();

function renderAgenteType() {
  const el = document.getElementById('agente-tipo-section');
  if (!el) return;
  const tipo  = G.tipoAgente||'nenhum';
  const agente = AGENTES[tipo];
  const nivelIdx = ['nenhum','amigo','profissional','super'].indexOf(tipo);

  let html = `<div class="agente-card">
    <div class="agente-header">
      <div class="agente-icon">${agente.icon}</div>
      <div><div class="agente-nome">${agente.nome}</div><div class="agente-sub">Comissão: ${Math.round(agente.comissao*100)}% · Contatos: ${agente.contatos}</div></div>
    </div>
    <div class="agente-level-bar"><div class="agente-level-fill" style="width:${(nivelIdx/3)*100}%"></div></div>
    <div style="font-size:11px;color:var(--text2)">${agente.desc}</div>
  </div>
  <div class="sec-lbl" style="margin-bottom:8px">Trocar de Agente</div>
  <div class="nivel-btns">`;

  Object.entries(AGENTES).forEach(([key, ag]) => {
    if (key==='nenhum') return;
    const active = tipo===key;
    const temGrana = G.dinheiro >= ag.custo;
    const onclk = active ? '' : `onclick="contratarAgente('${key}')"`;
    html += `<div class="nivel-btn ${active?'active':''}" ${onclk}>
      <div class="nb-icon">${ag.icon}</div>
      <div class="nb-info">
        <div class="nb-nome">${ag.nome} ${active?'<span style="color:var(--green);font-size:10px">✅ ATIVO</span>':''}</div>
        <div class="nb-desc">${ag.desc}</div>
      </div>
      ${active?'':('<div class="nb-custo">'+(temGrana?fmtMoney(ag.custo):'<span style="color:var(--red)">R$'+Math.round(ag.custo/1000)+'k</span>')+'</div>')}
    </div>`;
  });
  html += '</div>';
  el.innerHTML = html;
}

function renderAgenteObjetivo() {
  const el = document.getElementById('agente-objetivo-section');
  if (!el) return;
  const obj = G.objetivoAtivo;
  let html = '<div class="sec-lbl" style="margin-bottom:8px">🎯 Objetivo Ativo</div>';
  if (!obj || obj.status !== 'ativo') {
    html += '<div style="color:var(--text2);font-size:12px;padding:8px">Sem objetivo ativo. Consiga um agente e peça furos!</div>';
  } else {
    const pct = Math.min(100, Math.round(obj.tipo==='gols_sofridos'
      ? Math.max(0,(obj.meta-obj.progresso)/obj.meta*100)
      : obj.progresso/obj.meta*100));
    html += `<div class="objetivo-card">
      <div class="objetivo-title">🏆 ${obj.clubeInteressado} está de olho!</div>
      <div class="objetivo-desc">${obj.descricao}</div>
      <div class="objetivo-prog"><div class="objetivo-fill" style="width:${pct}%"></div></div>
      <div class="objetivo-nums">
        <span>${obj.progresso} / ${obj.meta}</span>
        <span>⏳ ${obj.prazoRestante} jogos restantes</span>
      </div>
    </div>`;
  }
  el.innerHTML = html;
}
