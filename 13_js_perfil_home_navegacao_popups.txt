function renderHomeScreen() {

  }
  // Mostrar card de talentos se player tiver arquétipo
  // Update agente info on home card
  const agInfoEl = document.getElementById('agente-home-info');
  if (agInfoEl) {
    const agt = AGENTES[G?.tipoAgente||'nenhum'];
    agInfoEl.textContent = G?.tipoAgente&&G.tipoAgente!=='nenhum'
      ? agt.icon+' '+agt.nome
      : 'Sem agente — contratar';
  }
  const cardTal = document.getElementById('card-talentos');
  if (cardTal) {
    const show = G.perfil==='jogador' && (G.arquetipo || (G.pontosArquetipo||0)>0 || G.jogos>=5);
    cardTal.style.display = show ? 'block' : 'none';
    const info = document.getElementById('talentos-info');
    if (info) {
      if (G.arquetipo) {
        const arq = getArquetipoPorGrupo(G.grupoPos).find(a=>a.key===G.arquetipo);
        info.textContent = (arq?arq.icon+' ':'')+( arq?arq.nome:'Arquétipo ativo')+' · ⭐'+(G.pontosArquetipo||0)+' pts';
      } else {
        info.textContent = 'Escolha seu arquétipo!';
        const badge = document.getElementById('talentos-badge');
        if (badge) badge.style.display='block';
      }
    }
  }
  document.getElementById('home-name').textContent  = G.nome.split(' ')[0];
  document.getElementById('season-title').textContent = G.clube;
  const posLiga = getPosicaoClubeTabela();
  document.getElementById('season-sub').textContent   = `${G.liga} - ${G.ano}` + (posLiga ? ` - ${posLiga}º` : '');
  document.getElementById('sg').textContent = G.gols;
  document.getElementById('sa').textContent = G.assists;

  document.getElementById('training-info').textContent = `Energia: ${G.energia}/${energiaCap()}`;
  const tabelaInfo = document.getElementById('tabela-home-info');
  if (tabelaInfo) {

function renderCalendario() {
  if (!G) return;
  const wrap = document.getElementById('cal-weeks');
  document.getElementById('cal-rodada').textContent = `Sem.${G.semanaAtual}`;
  const icons = {partida:'BOL',treino:'TRN',evento_vida:'VID',evento_midia:'MID',livre:'--'};
  wrap.innerHTML = G.calendario.map(c => {
    const janela = isJanelaTransferencia(c.semana);
    let cls = c.passada ? 'past' : '';
    if (!c.passada) {
      if (c.semana===G.semanaAtual) cls='current';
      else if (c.tipo==='partida')  cls='match';
      else if (c.tipo.startsWith('evento')) cls='event';
    }
    const mandoCampo = c.tipo === 'partida' ? (c.isHome ? 'C' : 'F') : '';
    const detalhe = mandoCampo + (janela ? 'J' : '');
    const title = c.tipo === 'partida'
      ? `Rodada ${c.rodada}: ${c.isHome?'casa':'fora'} vs ${c.adversario}${janela ? ' (Janela)' : ''}`
      : (janela ? 'Janela de transferencia' : c.tipo);
    return `<div class="cal-week ${cls}" title="${title}"><div class="cw-n">${c.semana}</div><div class="cw-icon">${icons[c.tipo]||'--'}</div><div style="font-size:8px;color:var(--text2);height:10px">${detalhe}</div></div>`;
  }).join('');
  const idx = G.calendario.findIndex(c=>c.semana===G.semanaAtual);
  if (idx>2) setTimeout(()=>{ wrap.scrollLeft=(idx-2)*56; },50);
}

function renderNoticias() {
  const list = document.getElementById('news-list');
  if (!G.noticias.length) { list.innerHTML='<div class="empty-notice">Nenhum registro ainda.</div>'; return; }
  list.innerHTML = G.noticias.slice(-8).reverse().map(n=>
    `<div class="news-item"><div class="news-icon">${n.icon}</div><div>
      <div class="news-headline">${n.texto}</div>
      <div class="news-time">Semana ${n.semana}, ${n.ano}</div>
    </div></div>`
  ).join('');
}

function contratarPatrocinioMidia(id) {
  if (!G) return;
  normalizarMidia();
  const p = PATROCINIOS_MIDIA.find(x => x.id === id);
  if (!p) return;
  if (G.midia.patrocinios.includes(id)) { showToast('Patrocinio ja ativo.'); return; }
  if ((G.fama||0) < p.famaReq) { showToast('Fama insuficiente.'); return; }
  G.midia.patrocinios.push(id);
  G.dinheiro += p.premio || 0;
  if (p.moral) G.moral = Math.min(100, G.moral + p.moral);
  if (p.fama) G.fama = Math.min(100, G.fama + p.fama);
  addNoticia(G,'PAT',G.nome+' assinou patrocinio: '+p.nome+'.');
  updateTopBar(); renderMidiaScreen(); saveGame(false);
  showToast('Patrocinio assinado: +'+fmtMoney(p.premio||0));
}

  const flag = nac?.label?.split(' ')[0] || '🌍';
  document.getElementById('top-name').textContent = G.nome;

  document.getElementById('top-age').textContent  = `${flag} ${G.idade}a · OVR ${G.ovr||'—'}`;

  document.getElementById('s-energia').textContent = G.energia + (_cap > 100 ? '/'+_cap : '');
  document.getElementById('s-moral').textContent   = G.moral;
  document.getElementById('s-fama').textContent    = G.fama;

}

function posLabel(key) {
  for (const g of Object.keys(POS_ESP)) {
    const f = POS_ESP[g].find(p=>p.key===key);
    if (f) return f.label;
  }
  return key||'—';
}

// Fila de popups: evita sobreposição quando múltiplos eventos disparam ao mesmo tempo
window._popupQueue = [];
window._popupAtivo = false;

function showPopup({ icon, titulo, corpo, botoes }) {
  // Se já há um popup aberto, enfileirar para exibir depois
  if (window._popupAtivo) {
    window._popupQueue.push({ icon, titulo, corpo, botoes });
    return;
  }
  _renderPopup({ icon, titulo, corpo, botoes });
}

function _renderPopup({ icon, titulo, corpo, botoes }) {
  window._popupAtivo = true;
  window._popupClickLocked = false;
  document.getElementById('pm-content').innerHTML = `
    <div class="pm-icon">${icon}</div>
    <div class="pm-title">${titulo}</div>
    <div class="pm-body">${corpo}</div>
    <div class="pm-choices">${botoes.map((b,i)=>`
      <button class="pm-btn" onclick="popupAction(${i})">
        <span class="ch-icon">${b.icon}</span>
        <span class="ch-text">${b.text||b.texto}</span>
        ${b.odds?`<span class="ch-odds">${b.odds}</span>`:''}
      </button>`).join('')}
    </div>`;
  window._ppBtns = botoes;
  document.getElementById('popup-overlay').classList.add('active');
}

function popupAction(i) {
  if (window._popupClickLocked) return;
  window._popupClickLocked = true;
  const acao = window._ppBtns?.[i]?.acao;
  if (acao) acao();
  else closePopup();
  setTimeout(() => {
    const ov = document.getElementById('popup-overlay');
    if (ov?.classList.contains('active')) window._popupClickLocked = false;
  }, 150);
}
function closePopup() {
  const ov = document.getElementById('popup-overlay');
  if (!ov) return;
  ov.classList.remove('active');
  window._ppBtns = null;
  window._popupAtivo = false;
  ov.onclick = closePopupOutside;
  // Exibir próximo popup da fila, se existir
  if (window._popupQueue.length > 0) {
    const proximo = window._popupQueue.shift();
    setTimeout(() => _renderPopup(proximo), 120);
  }
}
function closePopupOutside(e) { if(e.target===document.getElementById('popup-overlay')) closePopup(); }

        showToast('Saude +8, Energia +'+d+'!');

function renderInvestimentos() {
  if (!G) return;
  const grid = document.getElementById('loja-investimentos');
  if (!grid) return;
  const inv = G.inventario||{};
  grid.innerHTML = INVESTIMENTOS.map(item => {
    const comprado = !!inv[item.id];
    const podePagar = G.dinheiro >= item.preco;
    if (comprado) {
      return '<div class="loja-item" style="border-color:var(--green)">'+
        '<div class="li-icon">'+item.icon+'</div>'+
        '<div class="li-name">'+item.nome+' <span style="color:var(--green);font-size:10px">✅ ATIVO</span></div>'+
        '<div class="li-ef">'+item.efeitoLabel+'</div>'+
        '</div>';
    }
    return '<div class="loja-item '+(podePagar?'':'esgotado')+'" onclick="comprarInvestimento(\'' + item.id + '\')">'+
      '<div class="li-icon">'+item.icon+'</div>'+
      '<div class="li-name">'+item.nome+'</div>'+
      '<div class="li-desc">'+item.desc+'</div>'+
      '<div class="li-ef">'+item.efeitoLabel+'</div>'+
      '<div class="li-preco">'+fmtMoney(item.preco)+'</div>'+
      '</div>';
  }).join('');
}

function comprarInvestimento(id) {
  if (!G) return;
  const item = INVESTIMENTOS.find(i=>i.id===id);
  if (!item) return;
  // Guard: já possui este investimento
  if ((G.inventario||{})[item.id]) {
    showToast('✅ Já tens este investimento ativo!'); return;
  }
  if (G.dinheiro < item.preco) { showToast('💸 Dinheiro insuficiente!'); return; }
  // Guard extra: lock de clique duplo rápido
  if (comprarInvestimento._lock) return;
  comprarInvestimento._lock = true;
  setTimeout(() => { comprarInvestimento._lock = false; }, 1500);
  showPopup({
    icon:item.icon, titulo:'Investir em '+item.nome+'?',
    corpo:item.desc+'\n\n'+item.efeitoLabel+'\n\nInvestimento: '+fmtMoney(item.preco)+'\nSaldo atual: '+fmtMoney(G.dinheiro),
    botoes:[
      { icon:'💰', text:'Investir', acao:()=>{
        // BUG FIX: re-validar saldo no momento da confirmação
        if (G.dinheiro < item.preco) { showToast('💸 Saldo insuficiente!'); closePopup(); return; }
        if ((G.inventario||{})[item.id]) { showToast('✅ Já tens este investimento!'); closePopup(); return; }
        G.dinheiro -= item.preco;
        const inv = G.inventario||{};
        inv[item.id] = 1; G.inventario = inv;
        if (item.fama) G.fama = Math.min(100, G.fama + item.fama);
        closePopup(); renderLojaScreen(); updateTopBar(); saveGame(false);
        addNoticia(G,item.icon,G.nome+' investiu em '+item.nome+'! Renda passiva ativa.');
        showToast(item.icon+' '+item.nome+' ativo! Rende por semana.');
      }},
      { icon:'❌', text:'Cancelar', acao:closePopup }
    ]
  });
}
