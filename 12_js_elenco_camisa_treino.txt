function openLineupMenu() {
  if (!MS) return;
  document.getElementById('lineup-panel').style.display = 'flex';
  // Auto-pause
  if (MS.running && !MS.paused) togglePause();
  renderLineupContent(MS.lpTab||'home');
}
function closeLineupMenu() {
  document.getElementById('lineup-panel').style.display = 'none';
  if (MS && MS._halfTimeDone && MS.paused && !MS._intervaloResolvido && !MS.done) {
    setTimeout(abrirIntervaloPartida, 100);
  }
}
function lpTab(tab) {
  if (!MS) return;
  MS.lpTab = tab;
  document.querySelectorAll('.lp-tab').forEach(el => {
    el.classList.toggle('active', el.textContent.toLowerCase().includes(
      tab==='home'?'mandante':tab==='away'?'visitante':'substit'
    ));
  });
  renderLineupContent(tab);
}

function subIntervaloNpc(side, num) {
  if (!MS || !MS._halfTimeDone || MS.done || G.perfil !== 'treinador') return;
  const squad = side === 'home' ? MS.squadA : MS.squadB;
  const usedKey = side === 'home' ? 'subsUsedA' : 'subsUsedB';
  if (MS[usedKey] >= MS.subLimit) { showToast('Limite de substituicoes atingido.'); return; }
  const out = squad.find(p => p.num === num && p.inGame && !p.isPlayer);
  const reservas = squad.filter(p => !p.inGame && !p.subbedOut && !p.isPlayer);
  if (!out || !reservas.length) { showToast('Sem substituicao disponivel.'); return; }
  MS.pendingSub = { side, outNum:num };
  renderLineupContent(MS.lpTab || side);
}

function cancelarSubIntervaloNpc() {
  if (!MS) return;
  const side = MS.pendingSub?.side || MS.lpTab || 'home';
  MS.pendingSub = null;
  renderLineupContent(side);
}

function confirmarSubIntervaloNpc(side, outNum, inNum) {
  if (!MS || !MS._halfTimeDone || MS.done || G.perfil !== 'treinador') return;
  const squad = side === 'home' ? MS.squadA : MS.squadB;
  const usedKey = side === 'home' ? 'subsUsedA' : 'subsUsedB';
  if (MS[usedKey] >= MS.subLimit) { showToast('Limite de substituicoes atingido.'); cancelarSubIntervaloNpc(); return; }
  const out = squad.find(p => p.num === outNum && p.inGame && !p.isPlayer);
  const sub = squad.find(p => p.num === inNum && !p.inGame && !p.subbedOut && !p.isPlayer);
  if (!out || !sub) { showToast('Escolha de substituicao invalida.'); cancelarSubIntervaloNpc(); return; }
  out.inGame = false; out.subbedOut = true;
  sub.inGame = true; sub.energia = Math.min(100, sub.energia + 8);
  MS[usedKey]++;
  MS.pendingSub = null;
  addNpcRating(side === 'home' ? 'teamA' : 'teamB', sub.num, 0.15);
  logMatch('Substituicao no intervalo: camisa '+out.num+' sai, camisa '+sub.num+' entra.', 'hl');
  renderLineupContent(MS.lpTab || side);
}

function renderLineupContent(tab) {
  const content = document.getElementById('lp-content');
  if (!content || !MS) return;
  tab = tab || 'home';

  if (tab === 'subs') {
    const subsA = MS.subsUsedA, subsB = MS.subsUsedB, lim = MS.subLimit;
    let html = '<div class="lp-sub-info">🟢 '+MS.home+': '+subsA+'/'+lim+' subs  |  🔴 '+MS.away+': '+subsB+'/'+lim+' subs</div>';
    const bench = [...MS.squadA.filter(p=>!p.inGame), ...MS.squadB.filter(p=>!p.inGame)];
    if (!bench.length) { html += '<div class="lp-sub-info">Nenhum reserva disponível.</div>'; }
    bench.forEach(p => {
      const e = Math.round(p.energia);
      const bc = e>50?'var(--green)':e>25?'var(--gold)':'var(--red)';
      html += '<div class="lp-player'+(p.isPlayer?' is-player':'')+'">'+
        '<div class="lp-num">'+p.num+'</div>'+
        '<div class="lp-info"><div class="lp-name">'+(p.isPlayer?'⭐ ':'')+p.name+'</div>'+
        '<div class="lp-pos">'+p.pos+' | EN: '+e+'</div>'+
        '<div class="lp-enbar"><div class="lp-enfill" style="width:'+e+'%;background:'+bc+'"></div></div></div>'+
        '<div class="lp-ovr">'+p.ovr+'</div></div>';
    });
    content.innerHTML = html;
    return;
  }

  const squad    = tab==='home' ? MS.squadA : MS.squadB;
  const subsUsed = tab==='home' ? MS.subsUsedA : MS.subsUsedB;
  const lim      = MS.subLimit;
  let html = '<div class="lp-sub-info">Substituições usadas: '+subsUsed+'/'+lim+'</div>';
  const pending = MS.pendingSub && MS.pendingSub.side === tab ? MS.pendingSub : null;
  if (pending) {
    const outP = squad.find(p => p.num === pending.outNum);
    const reservas = squad.filter(p => !p.inGame && !p.subbedOut && !p.isPlayer);
    html += '<div class="lp-sub-info">Sai: camisa '+(outP?outP.num:'?')+' '+(outP?outP.name:'')+'. Escolha quem entra:</div>';
    if (!reservas.length) html += '<div class="lp-sub-info">Nenhum reserva disponivel.</div>';
    reservas.forEach(r => {
      html += '<div class="lp-player">'+
        '<div class="lp-num">'+r.num+'</div>'+
        '<div class="lp-info"><div class="lp-name">'+r.name+'</div>'+
        '<div class="lp-pos">'+r.pos+' | EN: '+Math.round(r.energia)+'</div></div>'+
        '<div class="lp-ovr">'+r.ovr+'</div>'+
        '<button class="talent-btn" onclick="confirmarSubIntervaloNpc(&quot;'+tab+'&quot;,'+pending.outNum+','+r.num+')">Entra</button>'+
        '</div>';
    });
    html += '<button class="talent-btn" onclick="cancelarSubIntervaloNpc()">Cancelar</button>';
  }
  squad.filter(p=>p.inGame || p.subbedOut).forEach(p => {
    const e   = Math.round(p.energia);
    const bc  = e>50?'var(--green)':e>25?'var(--gold)':'var(--red)';
    const warn = e<=25 && p.inGame ? '⚠️ ' : '';
    html += '<div class="lp-player'+(p.isPlayer?' is-player':'')+(p.subbedOut?' subbed-out':'')+(pending && pending.outNum===p.num?' is-player':'')+'">'+
      '<div class="lp-num">'+p.num+'</div>'+
      '<div class="lp-info">'+
      '<div class="lp-name">'+(p.isPlayer?'⭐ ':'')+p.name+(p.subbedOut?' 🔄':'')+'</div>'+
      '<div class="lp-pos">'+warn+p.pos+(p.inGame?' | EN: '+e:'')+'</div>'+
      (p.inGame?'<div class="lp-enbar"><div class="lp-enfill" style="width:'+e+'%;background:'+bc+'"></div></div>':'')+
      '</div><div class="lp-ovr">'+p.ovr+'</div>'+
      (G.perfil==='treinador' && MS._halfTimeDone && MS.paused && p.inGame && !p.isPlayer && subsUsed < lim && !pending
        ? '<button class="talent-btn" onclick="subIntervaloNpc(&quot;'+tab+'&quot;,'+p.num+')">Sub</button>'
        : '')+
      '</div>';
  });
  content.innerHTML = html;
}

function pedirCamisa() {
  if (!G) return;
  showPopup({
    icon:'👕', titulo:'Pedir Número de Camisa',
    corpo:'Qual número você quer usar?\n\n(O clube pode aceitar ou recusar conforme disponibilidade)\n\nCamisa atual: '+( G.camisa||10),
    botoes:[
      {icon:'🔟', text:'Camisa 10',  acao:()=>processarPedidoCamisa(10)},
      {icon:'9️⃣', text:'Camisa 9',   acao:()=>processarPedidoCamisa(9)},
      {icon:'7️⃣', text:'Camisa 7',   acao:()=>processarPedidoCamisa(7)},
      {icon:'✏️', text:'Outro número',acao:()=>{ closePopup(); pedirCamisaCustom(); }},
    ]
  });
}

function pedirCamisaCustom() {
  const num = parseInt(prompt('Digite o número da camisa (1-99):'), 10);
  if (!num || num<1||num>99) { showToast('❌ Número inválido.'); return; }
  processarPedidoCamisa(num);
}

function processarPedidoCamisa(num) {
  closePopup();
  if (!G) return;
  G.camisaPedida = num;
  // 70% de chance de conseguir se for número acima de 20
  // 50% se for cobiçado (7,9,10,11)
  const cobiçado = [7,9,10,11].includes(num);
  const chance = cobiçado ? 0.50 : 0.75;
  if (Math.random() < chance) {
    G.camisa = num;
    showToast('✅ O clube aceitou! Você agora usa a camisa '+num+'!');
    addNoticia(G,'👕',G.nome+' passa a usar a camisa '+num+' no '+G.clube+'.');
  } else {
    showToast('❌ O clube recusou. A camisa '+num+' não está disponível.');
    addNoticia(G,'👕',G.nome+' pediu a camisa '+num+', mas o clube recusou.');
  }
  saveGame(false);
  populateProfileScreen();
}

function renderSkillsSection() {

  const arqs = getArquetipoPorGrupo(G.grupoPos);
  const arqAtual = arqs.find(a => a.key === G.arquetipo);
  if (!arqAtual) { el.innerHTML = ''; return; }

  if ((G.ovr||0) < 60) {
    el.innerHTML = `<div style="background:var(--card);border:1.5px solid var(--border);border-radius:var(--radius);padding:16px;text-align:center;color:var(--text2);font-size:13px;">🔒 Skills desbloqueiam com OVR 60+ (atual: ${G.ovr||0})</div>`;
    return;
  }

  const talentosDisponiveis = arqAtual.talentos || [];
  const desbloqueados = G.talentos || [];
  const pts = G.pontosArquetipo || 0;

  const itens = talentosDisponiveis.map(t => {
    const tem = desbloqueados.includes(t.key);
    const podeComprar = !tem && pts >= t.custo;

function comprarTalento(key) {
  if (!G) return;
  const arqs = getArquetipoPorGrupo(G.grupoPos);
  const arqAtual = arqs.find(a => a.key === G.arquetipo);
  const t = arqAtual?.talentos?.find(t => t.key === key);
  if (!t) return;
  if ((G.talentos||[]).includes(key)) { showToast('✅ Já tens esta skill!'); return; }
  if ((G.pontosArquetipo||0) < t.custo) { showToast('⚠️ Pontos insuficientes!'); return; }
  G.talentos = [...(G.talentos||[]), key];
  G.pontosArquetipo = (G.pontosArquetipo||0) - t.custo;
  // Aplicar bónus de atributo
  Object.entries(t.attrBonus||{}).forEach(([attr, val]) => {
    addAttrBonus(attr, val);
  });
  G.ovr = calculateOverall(G);
  saveGame(false); updateTopBar();
  renderSkillsSection();
  showToast('🎯 Skill desbloqueada: '+t.nome+'!');
}

function trainAttr(key, label) {
  if (!G) return;
  if (G.perfil === 'treinador') {
    treinoColetivo(key);
    return;
  }

  // Limite: 3 treinos por dia
  const treinosHoje = G.treinoHoje || 0;
  if (treinosHoje >= 3) {
    showPopup({
      icon: '🚫', titulo: 'Limite diário atingido!',
      corpo: 'Você já treinou 3 vezes hoje.\nAvance para o próximo dia para continuar.\n\n💡 Dia de jogo não conta treinos — descanse bem!',
      botoes: [{ icon: '😴', text: 'Entendido', acao: closePopup }]
    });
    return;
  }

  // Checar energia mínima
  if (G.energia < 10) {
    showPopup({
      icon: '🔋', titulo: 'Sem energia!',
      corpo: `Energia: ${G.energia}/${energiaCap()}\nVocê precisa de pelo menos 10 de energia.\n\n💡 Use um energético (⚡ Estoque) ou avance o dia para recuperar.`,
      botoes: [
        { icon: '⚡', text: 'Usar Energético', acao: () => { closePopup(); abrirEstoqueRapido(); } },
        { icon: '❌', text: 'Fechar', acao: closePopup }
      ]
    });
    return;
  }

  // Verificar se está lesionado
  if (G.emLesao) {
    showToast('🩹 Você está lesionado! ('+(G.lesaoTipo||'lesão')+') — '+G.semanasLesao+' semanas para recuperar.');
    return;
  }

  // Custo FIXO por grupo de atributo — sem save-scumming
  // Físicos custam mais (corpo mais exigido), mentais custam menos
  const _attrDef = STATS_JOGADOR?.find?.(a => a.key === key)
                || STATS_GOLEIRO?.find?.(a => a.key === key);
  const _grupo   = _attrDef?.group || 'TEC';
  const _custos  = { FIS:17, ATK:15, TEC:13, DEF:13, GK:14, MEN:10 };
  const custo    = _custos[_grupo] || 14;
  const atual = G.attrs[key] || 0;
  // Auge: 24-29 anos = bônus dobrado. Declínio: 32+ = base 0
  const isAuge   = G.idade >= 24 && G.idade <= 29;
  const isDeclínio = G.idade >= 32;
  // MEN (mentais) não sofrem declínio — manter base >=1
  const isMental  = _grupo === 'MEN';
  const baseRaw   = G.idade <= 18 ? 3
                  : G.idade <= 23 ? 2
                  : isDeclínio ? (isMental ? 1 : 0)
                  : 1;
  const baseMult = isAuge ? 2 : 1;
  const base     = baseRaw * baseMult;
  G.auge = isAuge; // atualizar flag
  // No declínio, atributos físicos caem automaticamente (1 por temporada via fimDeTemporada)
  const gain  = Math.max(0, rng(base, base + 3) - Math.floor(atual / 25));
  // Campo de Treinamento: +1 extra por sessão
  const campoBonus = (G.inventario||{})['campo_treino'] ? 1 : 0;

  // Chance de lesão por treino intenso com energia baixa
  if (G.energia < 25 && Math.random() < 0.12) {
    aplicarLesao('treino');
    return;
  }

  G.energia = Math.max(0, G.energia - custo);
  G.attrs[key] = Math.min(99, atual + gain + campoBonus);
  G.treinoHoje = treinosHoje + 1;
  if (gain > 0) G.moral = Math.min(100, G.moral + 2);

  refreshOVR();

  // Pontos de talento: +1 a cada 5 OVR acima de 60
  if (G.ovr >= 60) {
    const ptsDevidos = Math.floor((G.ovr - 60) / 5);
    const gastos = (G.talentos||[]).reduce((s,k) => {
      const _a = getArquetipoPorGrupo(G.grupoPos).find(a=>a.key===G.arquetipo);
      return s + (_a?.talentos?.find(t=>t.key===k)?.custo||0);
    }, 0);
    G.pontosArquetipo = Math.max(G.pontosArquetipo||0, ptsDevidos - gastos);
  }

  updateTopBar(); updateEnergiaBar(); populateAttrsGrid(); saveGame(false);

  // Feedback do treino com contador do dia
  const treinosRestantes = 3 - G.treinoHoje;
  const pctAmanha = [20, 15, 10][Math.min(G.treinoHoje, 3) - 1] || 10;
  const augeInfo  = G.auge ? '\n⚡ PERÍODO DE AUGE! Treinos rendem o dobro!' : (G.idade>=32?'\n📉 Declínio físico — foque em atributos mentais.':'');
  const extraInfo = `\n\n💪 Treinos hoje: ${G.treinoHoje}/3 | Restam ${treinosRestantes}\n🌙 Recuperação amanhã: ~${pctAmanha}%${augeInfo}`;

  showPopup({
    icon: gain > 0 ? '📈' : '😓',
    titulo: gain > 0 ? 'Evolução!' : 'Sem evolução',
    corpo: gain > 0
      ? `${label} melhorou!\n${atual} → ${G.attrs[key]} (+${gain})\nOVR: ${G.ovr}${extraInfo}`
      : `Treino duro mas ${label} não evoluiu agora.${extraInfo}`,
    botoes: [{ icon: '✔️', text: 'Fechar', acao: closePopup }]
  });
}

// ─── PROFILE ─────────────────────────────────────────────────

function populateProfileScreen() {
  if (!G) return;
  const nac = NACIONALIDADES.find(n=>n.key===G.nacionalidade) || NACIONALIDADES[0];

  document.getElementById('profile-name').textContent  = G.nome;
  document.getElementById('profile-club').textContent  = `${G.clube} · ${G.liga}`;
  document.getElementById('profile-avatar').textContent = G.perfil==='treinador'?'🧢':'⚽';
  document.getElementById('top-avatar').textContent     = G.perfil==='treinador'?'🧢':'⚽';

  // Badges
  const badges=[];
  const agenteInfo = AGENTES[G.tipoAgente||'nenhum'];
  if(G.tipoAgente && G.tipoAgente!=='nenhum') badges.push({label:agenteInfo.icon+' '+agenteInfo.nome,cls:'gold'});
  if(G.tipoContrato==='base') badges.push({label:'🌱 Categoria Base',cls:''});
  if(G.fama>=50) badges.push({label:'⭐ Famoso',cls:'gold'});
  if((G.lideranca||5)>=15) badges.push({label:'👑 Líder Nato',cls:'gold'});
  if((G.compostura||5)>=15) badges.push({label:'🧘 Inabalável',cls:''});
  badges.push({label: nac.label, cls:''});
  document.getElementById('profile-badges').innerHTML=badges.map(b=>`<div class="badge ${b.cls}">${b.label}</div>`).join('');

  // Actualizar stats de liderança/compostura
  const lidEl = document.getElementById('prof-lideranca');
  const compEl = document.getElementById('prof-compostura');
  if (lidEl)  lidEl.textContent  = (G.lideranca||5)+'/20';
  if (compEl) compEl.textContent = (G.compostura||5)+'/20';

  // Summary stats (top grid)
  const ovr = G.ovr || calculateOverall(G);
  const ovrColor = ovr>=80?'#2dff6e':ovr>=65?'#f5c842':'#ff4757';
  const entros = G.entrosamento||30;
  const entrosColor = entros>=70?'#2dff6e':entros>=40?'#f5c842':'#ff4757';
  const items=[
    {label:'OVR',val:`<span style="color:${ovrColor};font-size:26px">${ovr}</span>`,big:true},
    {label:'Idade',val:G.idade+' anos'},{label:'Salário',val:fmtSalarioDual(G.salario)},
    {label:'Gols',val:G.gols},{label:'Assists',val:G.assists},{label:'Jogos',val:G.jogos},
    {label:'Entros.',val:`<span style="color:${entrosColor}">${entros}</span>`}
  ];
  document.getElementById('profile-stats').innerHTML=items.map(i=>
    `<div class="sf-card"><div class="sf-val">${i.val}</div><div class="sf-lbl">${i.label}</div></div>`
  ).join('');

  // Badges especiais: auge, lesão, arquétipo
  const extrasEl = document.getElementById('profile-extras');
  if (extrasEl && G.perfil==='jogador') {
    let extHtml = '';
    if (G.emLesao) extHtml += `<div class="lesao-badge">🩹 Lesionado: ${G.lesaoTipo} (${G.semanasLesao} sem.)</div>`;
    if (G.idade>=24&&G.idade<=29) extHtml += '<div class="auge-badge">⚡ PERÍODO DE AUGE — treinos rendem o dobro!</div>';
    if (G.idade>=32) extHtml += '<div class="lesao-badge">📉 Declínio físico gradual — foque em atributos mentais</div>';
    if (G.arquetipo) {
      const arq = getArquetipoPorGrupo(G.grupoPos).find(a=>a.key===G.arquetipo);
      if (arq) extHtml += `<div class="auge-badge">${arq.icon} ${arq.nome} | ⭐ ${G.pontosArquetipo||0} pts | ${(G.talentos||[]).length} talentos</div>`;
    }
    if (G.promessaEpoca&&!G.promessaEpoca.cumprida) extHtml += `<div class="promessa-card"><div class="promessa-title">🤝 Promessa à Diretoria</div><div class="promessa-body">${G.promessaEpoca.meta} — prazo: ${G.promessaEpoca.prazoAno}</div></div>`;
    extrasEl.innerHTML = extHtml;
  }

  // Full stats breakdown (only for jogador)
  const fullStatsEl = document.getElementById('profile-full-stats');
  if (!fullStatsEl) return;

  if (G.perfil !== 'jogador') {
    fullStatsEl.innerHTML = '';
    return;
  }

  const statList = G.grupoPos === 'GOL' ? STATS_GOLEIRO : STATS_JOGADOR;
  const groups = {};
  statList.forEach(s => {
    if (!groups[s.group]) groups[s.group] = [];
    groups[s.group].push(s);
  });

  const groupLabels = { ATK:'⚡ Ataque', TEC:'🎯 Técnico', FIS:'💪 Físico', MEN:'🧠 Mental', DEF:'🛡️ Defesa', GK:'🧤 Goleiro' };

  // Camisa do jogador no perfil
  const camisaSection = document.getElementById('profile-camisa-section');
  if (camisaSection && G.perfil==='jogador') {
    const numAtual = G.camisa || 10;
    const statusPedido = G.camisaPedida
      ? (G.camisa === G.camisaPedida ? '✅ Pedido atendido' : '⏳ Pedido pendente: camisa '+G.camisaPedida)
      : '';
    camisaSection.innerHTML =
      '<div class="sec-lbl" style="margin-top:12px;margin-bottom:8px">👕 Camisa</div>'+
      '<div class="camisa-row">'+
      '<div class="camisa-badge">'+numAtual+'</div>'+
      '<div class="camisa-info">'+
        '<div class="camisa-num">Camisa '+numAtual+'</div>'+
        (statusPedido?'<div class="camisa-status">'+statusPedido+'</div>':'')+
      '</div>'+
      '<button class="camisa-btn" onclick="pedirCamisa()">Pedir outra</button>'+
      '</div>';
  }

  fullStatsEl.innerHTML = Object.entries(groups).map(([grp, stats]) => `
    <div class="stats-group">
      <div class="stats-group-lbl">${groupLabels[grp]||grp}</div>
      ${stats.map(s => {
        const val = G.attrs[s.key]||0;
        const clr = val>=80?'#2dff6e':val>=65?'#f5c842':val>=50?'#5bceff':'#ff4757';
        return `<div class="stat-row-full">
          <span class="srf-icon">${s.icon}</span>
          <span class="srf-lbl">${s.label}</span>
          <div class="srf-bar"><div class="srf-fill" style="width:${val}%;background:${clr}"></div></div>
          <span class="srf-val" style="color:${clr}">${val}</span>
        </div>`;
      }).join('')}
    </div>
  `).join('');
}
